def create_token(amount: float, upi_id: str) -> str:
    payload = {"amount": amount, "upi_id": upi_id, "ts": now()}
    signature = hmac_sha256(payload, device_key)
    return qr_encode(payload, signature)  # scan-to-pay, no internet needed
